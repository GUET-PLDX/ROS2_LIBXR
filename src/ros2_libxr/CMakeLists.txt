cmake_minimum_required(VERSION 3.16)
project(ros2_libxr)
include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(ros_interfaces REQUIRED)
add_subdirectory(libxr)
add_executable(ros2_libxr_node src/ros2libxr.cpp)
target_include_directories(ros2_libxr_node PRIVATE include)
ament_target_dependencies(ros2_libxr_node rclcpp geometry_msgs nav_msgs ros_interfaces)
target_link_libraries(ros2_libxr_node xr)
if(BUILD_TESTING)
  add_executable(ros2_libxr_wire_test test/wire_test.cpp)
  target_include_directories(ros2_libxr_wire_test PRIVATE include)
  target_link_libraries(ros2_libxr_wire_test xr)
  add_test(NAME ros2_libxr_wire_test COMMAND ros2_libxr_wire_test)
endif()
install(TARGETS ros2_libxr_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_package()
