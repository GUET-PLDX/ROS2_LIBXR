cmake_minimum_required(VERSION 3.16)
project(ros2_libxr)
include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(ros_interfaces REQUIRED)
add_subdirectory(libxr)
add_executable(ros2_libxr_node src/ros2libxr.cpp)
target_include_directories(ros2_libxr_node PRIVATE include)
ament_target_dependencies(ros2_libxr_node rclcpp geometry_msgs nav_msgs ros_interfaces)
target_link_libraries(ros2_libxr_node xr)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(tunnel_command_test test/tunnel_command_test.cpp src/ros2libxr.cpp)
  target_compile_definitions(tunnel_command_test PRIVATE ROS2_LIBXR_NO_MAIN)
  target_include_directories(tunnel_command_test PRIVATE include)
  ament_target_dependencies(tunnel_command_test rclcpp geometry_msgs nav_msgs ros_interfaces)
  target_link_libraries(tunnel_command_test xr)
  set_tests_properties(tunnel_command_test PROPERTIES ENVIRONMENT "ROS_DOMAIN_ID=229")
  add_executable(ros2_libxr_wire_test test/wire_test.cpp)
  target_include_directories(ros2_libxr_wire_test PRIVATE include)
  target_link_libraries(ros2_libxr_wire_test xr)
  add_test(NAME ros2_libxr_wire_test COMMAND ros2_libxr_wire_test)
  add_executable(ros2_libxr_yaw_joint_test test/yaw_joint_test.cpp)
  target_include_directories(ros2_libxr_yaw_joint_test PRIVATE include)
  target_link_libraries(ros2_libxr_yaw_joint_test xr)
  add_test(NAME ros2_libxr_yaw_joint_test COMMAND ros2_libxr_yaw_joint_test)
endif()
install(TARGETS ros2_libxr_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_package()
